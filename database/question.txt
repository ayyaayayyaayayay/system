USE `naap_evaluation_system`;

START TRANSACTION;

SET @semester_slug := '1st-semester-2026-2027';
SET @semester_id := (
    SELECT `id`
    FROM `semesters`
    WHERE `slug` = @semester_slug
    LIMIT 1
);

INSERT INTO `questionnaires` (
    `semester_id`,
    `evaluation_type_id`,
    `title`,
    `description`,
    `status`
)
SELECT
    @semester_id AS `semester_id`,
    et.`id` AS `evaluation_type_id`,
    src.`title`,
    src.`description`,
    'published' AS `status`
FROM (
    SELECT
        'student-professor' AS `evaluation_type_code`,
        'Student to Professor Evaluation Form' AS `title`,
        'Please provide your honest feedback about your professors.' AS `description`
    UNION ALL
    SELECT
        'supervisor-professor',
        'Supervisor to Professor Evaluation Form',
        'Please provide your evaluation of the professor''s performance.'
    UNION ALL
    SELECT
        'professor-professor',
        'Peer to Peer Evaluation Form',
        'Please provide your professional assessment of your colleague.'
) AS src
JOIN `evaluation_types` et
    ON et.`code` = src.`evaluation_type_code`
WHERE @semester_id IS NOT NULL
ON DUPLICATE KEY UPDATE
    `title` = VALUES(`title`),
    `description` = VALUES(`description`),
    `status` = VALUES(`status`);

    ON qt.`code` = src.`question_type_code`
