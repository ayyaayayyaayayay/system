SET @semester_slug := _utf8mb4'1st-semester-2026-2027' COLLATE utf8mb4_unicode_ci;
SET @evaluation_type_code := _utf8mb4'student-professor' COLLATE utf8mb4_unicode_ci;
SET @professor_employee_id := _utf8mb4'PRF-2026-003' COLLATE utf8mb4_unicode_ci;
SET @subject_code := _utf8mb4'AVT101' COLLATE utf8mb4_unicode_ci;

SET @semester_id := (
    SELECT `id`
    FROM `semesters`
    WHERE `slug` = @semester_slug
    LIMIT 1
);

SET @evaluation_type_id := (
    SELECT `id`
    FROM `evaluation_types`
    WHERE `code` = @evaluation_type_code
    LIMIT 1
);

SET @questionnaire_id := (
    SELECT `id`
    FROM `questionnaires`
    WHERE `semester_id` = @semester_id
      AND `evaluation_type_id` = @evaluation_type_id
      AND `status` = 'published'
    LIMIT 1
);

SET @professor_id := (
    SELECT u.`id`
    FROM `users` u
    JOIN `roles` r
      ON r.`id` = u.`role_id`
     AND r.`code` = 'professor'
    JOIN `staff_profiles` sp
      ON sp.`user_id` = u.`id`
     AND sp.`employee_id` = @professor_employee_id
     AND sp.`is_active` = 1
     AND sp.`deleted_at` IS NULL
    WHERE u.`status` = 'active'
      AND u.`deleted_at` IS NULL
    LIMIT 1
);

-- Select the AVT101 offering containing this exact 40-student seed cohort.
SET @course_offering_id := (
    SELECT co.`id`
    FROM `course_offerings` co
    JOIN `subjects` subject_row
      ON subject_row.`id` = co.`subject_id`
     AND subject_row.`subject_code` = @subject_code
    JOIN `student_course_enrollments` sce
      ON sce.`course_offering_id` = co.`id`
     AND sce.`status` = 'enrolled'
    JOIN `student_profiles` student_profile
      ON student_profile.`user_id` = sce.`student_id`
     AND student_profile.`is_active` = 1
     AND student_profile.`deleted_at` IS NULL
    JOIN `tmp_alexa_evaluation_seed` seed
      ON seed.`student_number` = student_profile.`student_number`
    WHERE co.`semester_id` = @semester_id
      AND co.`professor_id` = @professor_id
      AND co.`is_active` = 1
      AND co.`deleted_at` IS NULL
    GROUP BY co.`id`
    HAVING COUNT(DISTINCT seed.`student_number`) = 40
    ORDER BY co.`id`
    LIMIT 1
);

-- Fail before permanent writes if the current data no longer matches this seed.
CREATE TEMPORARY TABLE `tmp_alexa_seed_context` (
    `semester_id` BIGINT UNSIGNED NOT NULL,
    `evaluation_type_id` SMALLINT UNSIGNED NOT NULL,
    `questionnaire_id` BIGINT UNSIGNED NOT NULL,
    `professor_id` BIGINT UNSIGNED NOT NULL,
    `course_offering_id` BIGINT UNSIGNED NOT NULL,
    `question_count` TINYINT UNSIGNED NOT NULL CHECK (`question_count` = 16),
    `rating_question_count` TINYINT UNSIGNED NOT NULL CHECK (`rating_question_count` = 15),
    `qualitative_question_count` TINYINT UNSIGNED NOT NULL CHECK (`qualitative_question_count` = 1),
    `enrolled_seed_count` TINYINT UNSIGNED NOT NULL CHECK (`enrolled_seed_count` = 40)
) ENGINE=InnoDB;

INSERT INTO `tmp_alexa_seed_context` (
    `semester_id`, `evaluation_type_id`, `questionnaire_id`,
    `professor_id`, `course_offering_id`, `question_count`,
    `rating_question_count`, `qualitative_question_count`, `enrolled_seed_count`
)
SELECT
    @semester_id,
    @evaluation_type_id,
    @questionnaire_id,
    @professor_id,
    @course_offering_id,
    COUNT(DISTINCT q.`id`),
    COUNT(DISTINCT CASE WHEN qt.`code` = 'rating' THEN q.`id` END),
    COUNT(DISTINCT CASE WHEN qt.`code` = 'qualitative' THEN q.`id` END),
    (
        SELECT COUNT(DISTINCT seed.`student_number`)
        FROM `tmp_alexa_evaluation_seed` seed
        JOIN `student_profiles` student_profile
          ON student_profile.`student_number` = seed.`student_number`
         AND student_profile.`is_active` = 1
         AND student_profile.`deleted_at` IS NULL
        JOIN `users` student_user
          ON student_user.`id` = student_profile.`user_id`
         AND student_user.`status` = 'active'
         AND student_user.`deleted_at` IS NULL
        JOIN `roles` student_role
          ON student_role.`id` = student_user.`role_id`
         AND student_role.`code` = 'student'
        JOIN `student_course_enrollments` enrollment
          ON enrollment.`student_id` = student_user.`id`
         AND enrollment.`course_offering_id` = @course_offering_id
         AND enrollment.`status` = 'enrolled'
    )
FROM `questions` q
JOIN `question_types` qt
  ON qt.`id` = q.`question_type_id`
WHERE q.`questionnaire_id` = @questionnaire_id
  AND q.`is_active` = 1
  AND q.`deleted_at` IS NULL;

START TRANSACTION;

INSERT INTO `evaluations` (
    `semester_id`,
    `questionnaire_id`,
    `evaluation_type_id`,
    `evaluator_user_id`,
    `evaluatee_user_id`,
    `course_offering_id`,
    `general_comments`,
    `submitted_at`,
    `status`,
    `submission_duplicate_key`
)
SELECT
    @semester_id,
    @questionnaire_id,
    @evaluation_type_id,
    student_user.`id`,
    @professor_id,
    @course_offering_id,
    NULL,
    seed.`submitted_at`,
    'submitted',
    SHA2(CONCAT(
        'evaluation-submission-v1|', @evaluation_type_code, '|',
        @semester_id, '|', @evaluation_type_id, '|', student_user.`id`,
        '|course|', @course_offering_id
    ), 256)
FROM `tmp_alexa_evaluation_seed` seed
JOIN `student_profiles` student_profile
  ON student_profile.`student_number` = seed.`student_number`
 AND student_profile.`is_active` = 1
 AND student_profile.`deleted_at` IS NULL
JOIN `users` student_user
  ON student_user.`id` = student_profile.`user_id`
 AND student_user.`status` = 'active'
 AND student_user.`deleted_at` IS NULL
JOIN `roles` student_role
  ON student_role.`id` = student_user.`role_id`
 AND student_role.`code` = 'student'
JOIN `student_course_enrollments` enrollment
  ON enrollment.`student_id` = student_user.`id`
 AND enrollment.`course_offering_id` = @course_offering_id
 AND enrollment.`status` = 'enrolled'
WHERE NOT EXISTS (
    SELECT 1
    FROM `evaluations` existing
    WHERE existing.`semester_id` = @semester_id
      AND existing.`evaluation_type_id` = @evaluation_type_id
      AND existing.`evaluator_user_id` = student_user.`id`
      AND existing.`course_offering_id` = @course_offering_id
)
ORDER BY seed.`seed_order`;

SET @inserted_evaluation_count := ROW_COUNT();

-- Map the 15 ratings and one comment to the current questionnaire by sort order.
INSERT INTO `evaluation_responses` (
    `evaluation_id`,
    `question_id`,
    `rating_value`,
    `text_value`,
    `display_order`
)
SELECT
    evaluation_row.`id`,
    question_row.`id`,
    CASE
        WHEN question_type.`code` <> 'rating' THEN NULL
        WHEN question_row.`sort_order` = 1  THEN seed.`q01`
        WHEN question_row.`sort_order` = 2  THEN seed.`q02`
        WHEN question_row.`sort_order` = 3  THEN seed.`q03`
        WHEN question_row.`sort_order` = 4  THEN seed.`q04`
        WHEN question_row.`sort_order` = 5  THEN seed.`q05`
        WHEN question_row.`sort_order` = 6  THEN seed.`q06`
        WHEN question_row.`sort_order` = 7  THEN seed.`q07`
        WHEN question_row.`sort_order` = 8  THEN seed.`q08`
        WHEN question_row.`sort_order` = 9  THEN seed.`q09`
        WHEN question_row.`sort_order` = 10 THEN seed.`q10`
        WHEN question_row.`sort_order` = 11 THEN seed.`q11`
        WHEN question_row.`sort_order` = 12 THEN seed.`q12`
        WHEN question_row.`sort_order` = 13 THEN seed.`q13`
        WHEN question_row.`sort_order` = 14 THEN seed.`q14`
        WHEN question_row.`sort_order` = 15 THEN seed.`q15`
        ELSE NULL
    END,
    CASE
        WHEN question_type.`code` = 'qualitative' THEN seed.`comment_text`
        ELSE NULL
    END,
    question_row.`sort_order`
FROM `tmp_alexa_evaluation_seed` seed
JOIN `student_profiles` student_profile
  ON student_profile.`student_number` = seed.`student_number`
JOIN `evaluations` evaluation_row
  ON evaluation_row.`semester_id` = @semester_id
 AND evaluation_row.`questionnaire_id` = @questionnaire_id
 AND evaluation_row.`evaluation_type_id` = @evaluation_type_id
 AND evaluation_row.`evaluator_user_id` = student_profile.`user_id`
 AND evaluation_row.`evaluatee_user_id` = @professor_id
 AND evaluation_row.`course_offering_id` = @course_offering_id
 AND evaluation_row.`submitted_at` = seed.`submitted_at`
 AND evaluation_row.`submission_duplicate_key` = SHA2(CONCAT(
        'evaluation-submission-v1|', @evaluation_type_code, '|',
        @semester_id, '|', @evaluation_type_id, '|', student_profile.`user_id`,
        '|course|', @course_offering_id
    ), 256)
JOIN `questions` question_row
  ON question_row.`questionnaire_id` = @questionnaire_id
 AND question_row.`is_active` = 1
 AND question_row.`deleted_at` IS NULL
JOIN `question_types` question_type
  ON question_type.`id` = question_row.`question_type_id`
WHERE question_row.`sort_order` BETWEEN 1 AND 16
  AND NOT EXISTS (
      SELECT 1
      FROM `evaluation_responses` existing_response
      WHERE existing_response.`evaluation_id` = evaluation_row.`id`
        AND existing_response.`question_id` = question_row.`id`
  )
ORDER BY evaluation_row.`id`, question_row.`sort_order`;

SET @inserted_response_count := ROW_COUNT();

COMMIT;

DROP TEMPORARY TABLE `tmp_alexa_seed_context`;
DROP TEMPORARY TABLE `tmp_alexa_evaluation_seed`;

SELECT
    @semester_id AS `semester_id`,
    @questionnaire_id AS `questionnaire_id`,
    @professor_id AS `professor_id`,
    @course_offering_id AS `course_offering_id`,
    @inserted_evaluation_count AS `inserted_evaluations`,
    @inserted_response_count AS `inserted_responses`;
